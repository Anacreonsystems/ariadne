from __future__ import annotations

import hashlib
import json
from pathlib import Path

ROOT=Path('/workspace/ariadne')
EV=ROOT/'architecture/streaming-etl/evidence'
PROFILES=['flow-http','code','native-http','store-api','migrations','sqlite','legacy-contract','recorder','kernel','client','flow-api','task-api','flow-engine','flow-policy']

def artifact(name):
    return {'path':name,'sha256':hashlib.sha256((EV/name).read_bytes()).hexdigest()}

freeze=json.loads((EV/'native-stage5c-source-freeze-2026-11-03.json').read_text())
source=freeze['source_sha256']
assert len(source)==69 and freeze['verdict']=='PASS'
profiles={};totals={key:0 for key in ['tests','passed','failure','error','skipped']}
for profile in PROFILES:
    label='successor2' if profile=='flow-policy' else 'successor'
    name=f'native-stage5c-{profile}-{label}-2026-11-03.json'
    data=json.loads((EV/name).read_text())
    assert data['verdict']=='PASS',name
    assert data['source_sha256_before']==data['source_sha256_after']==source,name
    assert not data['source_drift'] and not data['source_inventory_drift'],name
    assert not data['unexpected_skip_or_xfail'],name
    for kind in ['log','junit','expected_markers']:
        assert artifact(data[kind]['path'])==data[kind],(name,kind)
    for key in totals:totals[key]+=data[key]
    profiles[profile]={'ledger':artifact(name),**{key:data[key] for key in totals},'duration_seconds':data['duration_seconds']}
for name,expected in [('flow-http',17),('code',7),('native-http',197)]:
    assert profiles[name]['tests']==expected,(name,profiles[name]['tests'])
assert all(hashlib.sha256((ROOT/path).read_bytes()).hexdigest()==digest for path,digest in source.items())
principal=json.loads((EV/'native-stage5c-principal-2026-11-03.json').read_text())
assert principal['verdict']=='PASS_N1_5C_BOUNDED',principal['verdict']
assert not principal.get('conditions_remaining',[])
static=json.loads((EV/'native-stage5c-static-2026-11-03.json').read_text())
assert static['verdict']=='PASS' and static['source_sha256']==source
concurrency=json.loads((EV/'native-stage5c-ordinary-concurrency-2026-11-03.json').read_text())
assert concurrency['verdict']=='PASS' and concurrency['tests']==1
assert concurrency['source_sha256_before']==concurrency['source_sha256_after']==source
for item in concurrency['artifacts']:
    assert artifact(item['path'])==item
