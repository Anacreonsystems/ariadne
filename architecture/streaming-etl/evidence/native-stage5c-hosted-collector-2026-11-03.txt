from collections import Counter
from datetime import datetime, timezone
import hashlib
import json
from pathlib import Path
import subprocess

root=Path('/workspace/ariadne');ev=root/'architecture/streaming-etl/evidence'
pr=json.loads(subprocess.check_output(['gh','pr','view','9','--repo','Anacreonsystems/ariadne','--json','url,state,headRefOid,mergeCommit,mergedAt,statusCheckRollup'],text=True))
assert pr['state']=='MERGED'
checks=pr['statusCheckRollup']
assert checks and all(x.get('status')=='COMPLETED' and x.get('conclusion') in ('SUCCESS','SKIPPED','NEUTRAL') for x in checks)
head,merge=pr['headRefOid'],pr['mergeCommit']['oid']
subprocess.run(['git','fetch','origin','main'],check=True,stdout=subprocess.DEVNULL)
read=lambda rev:subprocess.check_output(['git','rev-parse',rev],text=True).strip()
assert read(head+'^{tree}')==read(merge+'^{tree}')
acceptance=json.loads((ev/'native-stage5c-acceptance-2026-11-03.json').read_text())
identity=json.loads((ev/'native-stage5c-identity-python312-2026-11-03.json').read_text())
review=json.loads((ev/'native-stage5c-identity-principal-2026-11-03.json').read_text())
assert identity['verdict']=='PASS' and review['verdict']=='PASS_TEST_ONLY_IDENTITY_CORRECTION_SOURCE_QUALIFIED' and not review['conditions_remaining']
source=identity['source_sha256_after']
assert identity['source_sha256_before']==source
assert [p for p in source if source[p]!=acceptance['source_sha256'][p]]==['tests/test_native_stream_code.py']
for path,digest in source.items():
    for rev in (head,merge):
        assert hashlib.sha256(subprocess.check_output(['git','show',rev+':'+path])).hexdigest()==digest
